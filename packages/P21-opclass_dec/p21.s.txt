# P21 program - palindrome checker
# Rules: start at the label p21_main and finish with ret; every label you
# create starts with p21_ ; save ra (and any s registers you use) on the stack.
# You may call the library routines in sw/lib/rcv_lib.s (rcv_puts, rcv_getc, ...).


.data
p21_str:
    .asciz "racecar"

p21_msg_yes:
    .asciz "Palindrome\n"

p21_msg_no:
    .asciz "Not a palindrome\n"

.text
.globl main

main:
p21_main:
    # Save registers
    addi sp, sp, -12
    sw ra, 8(sp)
    sw s0, 4(sp)
    sw s1, 0(sp)

    # Initialize pointers
    la s0, p21_str
    mv s1, s0

# Find string length
p21_len:
    lbu t0, 0(s1)
    beqz t0, p21_len_done
    addi s1, s1, 1
    j p21_len

p21_len_done:
    addi s1, s1, -1

# Compare characters
p21_cmp:
    bgeu s0, s1, p21_yes

    lbu t0, 0(s0)
    lbu t1, 0(s1)

    bne t0, t1, p21_no

    addi s0, s0, 1
    addi s1, s1, -1
    j p21_cmp

# Palindrome result
p21_yes:
    la a0, p21_msg_yes
    j p21_print

# Not palindrome result
p21_no:
    la a0, p21_msg_no

# Print result
p21_print:
    li a7, 4
    ecall

    # Restore registers
    lw ra, 8(sp)
    lw s0, 4(sp)
    lw s1, 0(sp)
    addi sp, sp, 12

    # Exit program
    li a7, 10
    ecall
